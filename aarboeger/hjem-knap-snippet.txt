3) JavaScript - indsæt lige før </body>:

<script>
/* --- Hjem-knappen: i "auto" vises den kun, når filen er del af noget større --- */
(function(){
  var bar = document.getElementById("topbar");
  if (!bar) return;
  var link = document.getElementById("homeBtn");
  var mode = bar.getAttribute("data-mode") || "auto";
  var target = bar.getAttribute("data-href") || "index.html";
  function show(){ bar.classList.add("show"); }
  if (mode === "never") return;
  var fromUrl = null;
  try { var p = new URLSearchParams(location.search); fromUrl = p.get("hjem") || p.get("home"); } catch(e){}
  if (fromUrl) target = fromUrl;
  link.setAttribute("href", target);
  if (mode === "always" || fromUrl){ show(); return; }
  var path = (location.pathname || "").toLowerCase();
  var isIndex = path.charAt(path.length-1) === "/" || path.indexOf("/index.html") === path.length - 11;
  if (isIndex) return;
  try {
    if (window.matchMedia && (matchMedia("(display-mode: standalone)").matches || matchMedia("(display-mode: fullscreen)").matches || matchMedia("(display-mode: minimal-ui)").matches)){ show(); return; }
  } catch(e){}
  if (navigator.standalone === true){ show(); return; }
  try {
    if (document.referrer){
      var ref = new URL(document.referrer), here = new URL(location.href);
      var refDir = ref.pathname.slice(0, ref.pathname.lastIndexOf("/"));
      var hereDir = here.pathname.slice(0, here.pathname.lastIndexOf("/"));
      if (ref.origin === here.origin && refDir === hereDir && ref.href !== here.href){ show(); return; }
    }
  } catch(e){}
  if (location.protocol === "http:" || location.protocol === "https:"){
    try { fetch(target, {method:"HEAD"}).then(function(r){ if (r && r.ok) show(); }).catch(function(){}); } catch(e){}
  }
})();
</script>
